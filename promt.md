SELECT DISTINCT [PMName], [PM Number]
FROM dbo.[01_DATA_01_Data Mart Trial]
WHERE LTRIM(RTRIM([PMName])) IN (
    'MARLA G HOWELL','AUDRA J BURRIS','BURRIS, AUDRA J',
    'SHANTESHIA M NORMAN','NORMAN, SHANTESHIA M',
    'SUSAN F HOLMES','HOLMES, SUSAN F','TERENCE J DOLCH'
)

SELECT DISTINCT '[' + [PMName] + ']' AS PMName_WithBrackets, [PM Number]
FROM dbo.[01_DATA_01_Data Mart Trial] WITH (NOLOCK)
WHERE [PMName] LIKE '%HOWELL%'
   OR [PMName] LIKE '%BURRIS%'
   OR [PMName] LIKE '%NORMAN%'
   OR [PMName] LIKE '%HOLMES%'
   OR [PMName] LIKE '%DOLCH%'
ORDER BY PMName_WithBrackets;

SELECT 
  COUNT(*) AS TotalRows,
  COUNT([PMName]) AS HasName,
  COUNT([PM Number]) AS HasNumber,
  SUM(CASE WHEN [PMName] IS NOT NULL AND [PM Number] IS NULL THEN 1 ELSE 0 END) AS NameButNoNumber
FROM dbo.[01_DATA_01_Data Mart Trial] WITH (NOLOCK);
